function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetName = data.periodLabel || data.periodKey || "Untitled"; var sheet = ss.getSheetByName(sheetName); if (sheet) { sheet.clear(); sheet.clearFormats(); } else { sheet = ss.insertSheet(sheetName); } var row = 1; data.records.forEach(function (rec) { row = writeRecordBlock(sheet, row, data, rec); row += 2; }); sheet.setColumnWidth(1, 40); sheet.setColumnWidth(2, 420); sheet.setColumnWidth(3, 40); sheet.setColumnWidth(4, 40); sheet.setColumnWidth(5, 40); sheet.setColumnWidth(6, 150); sheet.setColumnWidth(7, 150); saveRawData(ss, data.periodLabel, JSON.stringify(data)); return ContentService.createTextOutput(JSON.stringify({ status: "ok", sheet: sheetName })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(err) })) .setMimeType(ContentService.MimeType.JSON); } } // Stores the exact JSON that was synced, keyed by period, in a hidden // "_rawdata" sheet — this is what lets the form READ its data back later // (e.g. after closing and reopening the tab), not just write to Sheets. function saveRawData(ss, periodLabel, json) { var sheet = ss.getSheetByName("_rawdata"); if (!sheet) { sheet = ss.insertSheet("_rawdata"); try { sheet.hideSheet(); } catch (e) {} sheet.getRange(1, 1, 1, 3).setValues([["periodLabel", "json", "updatedAt"]]); } // Force column A to plain text — otherwise Sheets silently auto-converts // a value like "August 2026" into an actual Date, which then never // string-matches the plain "August 2026" a GET request compares against. sheet.getRange("A:A").setNumberFormat("@"); var values = sheet.getDataRange().getValues(); var foundRow = -1; for (var i = 1; i < values.length; i++) { if (normalizePeriodCell_(values[i][0]) === periodLabel) { foundRow = i + 1; break; } } var now = new Date().toISOString(); if (foundRow > 0) { sheet.getRange(foundRow, 1).setValue(periodLabel); sheet.getRange(foundRow, 2).setValue(json); sheet.getRange(foundRow, 3).setValue(now); } else { sheet.appendRow([periodLabel, json, now]); } } // Handles rows that were already corrupted into a Date value before the // plain-text fix above existed, so old synced data still loads correctly. function normalizePeriodCell_(v) { if (Object.prototype.toString.call(v) === "[object Date]") { return Utilities.formatDate(v, Session.getScriptTimeZone(), "MMMM yyyy"); } return String(v); } // GET ?period= — lets the form load back whatever was last // synced for that month, so closing and reopening the tab doesn't lose it. function doGet(e) { try { var period = e.parameter && e.parameter.period; if (!period) { return ContentService.createTextOutput(JSON.stringify({ status: "ok", message: "GMP Audit sync endpoint is running." })) .setMimeType(ContentService.MimeType.JSON); } var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("_rawdata"); if (sheet) { var values = sheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (normalizePeriodCell_(values[i][0]) === period) { return ContentService.createTextOutput(JSON.stringify({ status: "ok", data: JSON.parse(values[i][1]) })) .setMimeType(ContentService.MimeType.JSON); } } } return ContentService.createTextOutput(JSON.stringify({ status: "not_found" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(err) })) .setMimeType(ContentService.MimeType.JSON); } } // Writes one record (a week, or the monthly summary) starting at startRow, // laid out the same way as the paper form: header table, findings columns // (0/1/2) in their own position per item, Score/Total/Rating rows, comments, // and signatures. Returns the next free row so blocks can stack. function writeRecordBlock(sheet, startRow, data, rec) { var r = startRow; var COLS = 7; var hdrStart = r; sheet.getRange(r, 1, 2, 4).merge().setValue(data.company).setFontWeight("bold").setFontSize(12); sheet.getRange(r, 5).setValue("Form No.:").setFontWeight("bold"); sheet.getRange(r, 6, 1, 2).merge().setValue(data.formNo); sheet.getRange(r + 1, 5).setValue("Issue No.:").setFontWeight("bold"); sheet.getRange(r + 1, 6, 1, 2).merge().setValue(data.issueNo); r += 2; sheet.getRange(r, 1, 2, 4).merge().setValue(data.formTitle).setFontWeight("bold").setFontSize(12); sheet.getRange(r, 5).setValue("Issue Date:").setFontWeight("bold"); sheet.getRange(r, 6, 1, 2).merge().setValue(data.issueDate); sheet.getRange(r + 1, 5).setValue("Record:").setFontWeight("bold"); sheet.getRange(r + 1, 6, 1, 2).merge().setValue(rec.label); r += 2; gridBorder(sheet.getRange(hdrStart, 1, r - hdrStart, COLS)); r += 1; if (rec.date) { sheet.getRange(r, 1).setValue("Date:").setFontWeight("bold"); sheet.getRange(r, 2).setValue(rec.date); sheet.getRange(r, 4, 1, 4).merge().setValue("(Frequency: Monthly — " + rec.label + ")").setFontStyle("italic").setFontSize(9); } if (rec.period) { sheet.getRange(r, 1).setValue("Period:").setFontWeight("bold"); sheet.getRange(r, 2).setValue(rec.period); sheet.getRange(r, 4, 1, 4).merge().setValue("(" + rec.label + ")").setFontStyle("italic").setFontSize(9); } r += 1; sheet.getRange(r, 1, 1, COLS).merge() .setValue("Definition: 0 - Not comply | 1 - Partially comply | 2 - Fully comply") .setFontStyle("italic").setFontSize(9); r += 2; rec.sections.forEach(function (section) { r = writeSectionTable(sheet, r, section); }); var totalRow = r; sheet.getRange(r, 1, 1, 2).merge().setValue("Total score (External + Internal):").setFontWeight("bold").setHorizontalAlignment("right"); sheet.getRange(r, 3, 1, 3).merge().setValue(rec.total).setHorizontalAlignment("center").setFontWeight("bold"); gridBorder(sheet.getRange(r, 1, 1, COLS)); r += 1; var ratingStart = r; var bands = [ { label: "Poor : 0-20 (take action)", match: "Poor" }, { label: "Good : 21-40 (need improvement)", match: "Good" }, { label: "Excellent : 41-72 (maintain)", match: "Excellent" } ]; bands.forEach(function (b) { var active = rec.ratingLabel && rec.ratingLabel.indexOf(b.match) === 0; var labelRange = sheet.getRange(r, 1, 1, 2).merge().setValue(b.label); var markRange = sheet.getRange(r, 3, 1, 3).merge(); if (active) { labelRange.setBackground("#e4ede2").setFontWeight("bold"); markRange.setValue("✓ Current").setBackground("#e4ede2").setFontWeight("bold").setHorizontalAlignment("center"); } r += 1; }); gridBorder(sheet.getRange(ratingStart, 1, 3, COLS)); r += 1; var commentsStart = r; if (rec.comments1 !== undefined) { sheet.getRange(r, 1).setValue("Comments (Outside):").setFontWeight("bold"); sheet.getRange(r, 2, 1, COLS - 1).merge().setValue(rec.comments1 || ""); r += 1; } if (rec.comments2 !== undefined) { sheet.getRange(r, 1).setValue("Comments (Internal):").setFontWeight("bold"); sheet.getRange(r, 2, 1, COLS - 1).merge().setValue(rec.comments2 || ""); r += 1; } if (rec.comments !== undefined) { sheet.getRange(r, 1).setValue("Comments:").setFontWeight("bold"); sheet.getRange(r, 2, 1, COLS - 1).merge().setValue(rec.comments || ""); r += 1; } if (r > commentsStart) gridBorder(sheet.getRange(commentsStart, 1, r - commentsStart, COLS)); r += 1; var sigStart = r; sheet.getRange(r, 1, 1, 4).merge().setValue(rec.signLeftLabel).setFontWeight("bold").setBackground("#e7e2d6"); sheet.getRange(r, 5, 1, 3).merge().setValue(rec.signRightLabel).setFontWeight("bold").setBackground("#e7e2d6"); r += 1; ["name", "position", "date"].forEach(function (f) { var lbl = f.charAt(0).toUpperCase() + f.slice(1) + ":"; sheet.getRange(r, 1).setValue(lbl).setFontWeight("bold"); sheet.getRange(r, 2, 1, 3).merge().setValue(rec.signLeft[f] || ""); sheet.getRange(r, 5).setValue(lbl).setFontWeight("bold"); sheet.getRange(r, 6, 1, 2).merge().setValue(rec.signRight[f] || ""); r += 1; }); gridBorder(sheet.getRange(sigStart, 1, r - sigStart, COLS)); if (rec.savedAt) { sheet.getRange(r, 1, 1, COLS).merge().setValue("Saved: " + rec.savedAt).setFontStyle("italic").setFontSize(8).setFontColor("#888888"); r += 1; } return r; } // Applies one clean full grid (outer box + every inner line) in a single // call, instead of bordering row-by-row — avoids the doubled-up / missing // lines that come from setting borders piecemeal on overlapping ranges. function gridBorder(range) { range.setBorder(true, true, true, true, true, true, "#000000", SpreadsheetApp.BorderStyle.SOLID); } // Writes one section (Outside/Internal perimeter) table: same column // positions as the form — #, Checking Criteria, then 0/1/2 as three // separate columns with the mark placed in whichever one applies. function writeSectionTable(sheet, startRow, section) { var r = startRow; var headStart = r; // Row 1: # | Checking Criteria | Findings (merged over 0/1/2) | When to complete | Person in charge sheet.getRange(r, 1, 2, 1).merge().setValue("#").setFontWeight("bold").setHorizontalAlignment("center").setVerticalAlignment("middle"); sheet.getRange(r, 2, 2, 1).merge().setValue("Checking Criteria").setFontWeight("bold").setVerticalAlignment("middle"); sheet.getRange(r, 3, 1, 3).merge().setValue("Findings").setFontWeight("bold").setHorizontalAlignment("center"); sheet.getRange(r, 6, 2, 1).merge().setValue("When to complete").setFontWeight("bold").setVerticalAlignment("middle"); sheet.getRange(r, 7, 2, 1).merge().setValue("Person in charge").setFontWeight("bold").setVerticalAlignment("middle"); // Row 2: 0 | 1 | 2 as their own columns, directly under "Findings" — same // three physical positions the mark gets written into below. sheet.getRange(r + 1, 3).setValue(0).setFontWeight("bold").setHorizontalAlignment("center"); sheet.getRange(r + 1, 4).setValue(1).setFontWeight("bold").setHorizontalAlignment("center"); sheet.getRange(r + 1, 5).setValue(2).setFontWeight("bold").setHorizontalAlignment("center"); sheet.getRange(r, 1, 2, 7).setBackground("#e7e2d6"); gridBorder(sheet.getRange(r, 1, 2, 7)); r += 2; sheet.getRange(r, 1, 1, 7).merge().setValue(section.name).setFontWeight("bold").setFontColor("#ffffff").setBackground("#3f5747"); gridBorder(sheet.getRange(r, 1, 1, 7)); r += 1; var itemsStart = r; section.items.forEach(function (item) { sheet.getRange(r, 1).setValue(item.no); sheet.getRange(r, 2).setValue(item.criteria); [0, 1, 2].forEach(function (v, i) { if (item.mark === v) { sheet.getRange(r, 3 + i).setValue(v).setFontWeight("bold").setBackground("#e4ede2"); } }); sheet.getRange(r, 6).setValue(item.when || ""); sheet.getRange(r, 7).setValue(item.person || ""); r += 1; }); gridBorder(sheet.getRange(itemsStart, 1, section.items.length, 7)); sheet.getRange(r, 1, 1, 2).merge().setValue("Score :").setFontWeight("bold").setHorizontalAlignment("right"); sheet.getRange(r, 3, 1, 3).merge().setValue(section.score).setHorizontalAlignment("center").setFontWeight("bold"); gridBorder(sheet.getRange(r, 1, 1, 7)); r += 2; return r; }
